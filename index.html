    <script>
        // Tables de correspondance Unicode pour le style Outline / Double-struck (Blackboard Bold)
        const upperMap = {
            'A': '𝔸', 'B': '𝔹', 'C': 'ℂ', 'D': '𝔻', 'E': '𝔼', 'F': '𝔽', 'G': '𝔾', 'H': 'ℍ', 
            'I': '𝕀', 'J': '𝕁', 'K': '𝕂', 'L': '𝕃', 'M': '𝕄', 'N': 'ℕ', 'O': '𝕆', 'P': 'ℙ', 
            'Q': 'ℚ', 'R': 'ℝ', 'S': '𝕊', 'T': '𝕋', 'U': '𝕌', 'V': '𝕍', 'W': '𝕎', 'X': '𝕏', 
            'Y': '𝕐', 'Z': 'ℤ'
        };

        const lowerMap = {
            'a': '𝕒', 'b': '𝕓', 'c': '𝕔', 'd': '𝕕', 'e': '𝕖', 'f': '𝕗', 'g': '𝕘', 'h': '𝕙', 
            'i': '𝕚', 'j': '𝕛', 'k': '𝕜', 'l': '𝕝', 'm': '𝕞', 'n': '𝕟', 'o': '𝕠', 'p': '𝕡', 
            'q': '𝕢', 'r': '𝕣', 's': '𝕤', 't': '𝕥', 'u': '𝕦', 'v': '𝕧', 'w': '𝕨', 'x': '𝕩', 
            'y': '𝕪', 'z': '𝕫'
        };

        const digitMap = {
            '0': '𝟘', '1': '𝟙', '2': '𝟚', '3': '𝟛', '4': '𝟜', '5': '𝟝', '6': '𝟞', '7': '𝟟', '8': '𝟠', '9': '𝟡'
        };

        // Fonction de conversion
        function convertToOutline(text) {
            let result = '';
            for (let i = 0; i < text.length; i++) {
                const char = text[i];
                if (upperMap[char]) {
                    result += upperMap[char];
                } else if (lowerMap[char]) {
                    result += lowerMap[char];
                } else if (digitMap[char]) {
                    result += digitMap[char];
                } else {
                    result += char; // Conserve les autres caractères (espaces, ponctuation, accents) tels quels
                }
            }
            return result;
        }

        const inputText = document.getElementById('inputText');
        const outputText = document.getElementById('outputText');
        const charCount = document.getElementById('charCount');
        const clearBtn = document.getElementById('clearBtn');
        const copyBtn = document.getElementById('copyBtn');
        const copyBtnText = document.getElementById('copyBtnText');
        const toast = document.getElementById('toast');
        const presetBtns = document.querySelectorAll('.preset-btn');

        // Événement de saisie en temps réel
        inputText.addEventListener('input', () => {
            const text = inputText.value;
            outputText.value = convertToOutline(text);
            charCount.textContent = `${text.length} caractère${text.length > 1 ? 's' : ''}`;
        });

        // Bouton effacer
        clearBtn.addEventListener('click', () => {
            inputText.value = '';
            outputText.value = '';
            charCount.textContent = '0 caractères';
            inputText.focus();
        });

        // Bouton de copie rapide avec méthode de compatibilité iFrame (document.execCommand)
        copyBtn.addEventListener('click', () => {
            const textToCopy = outputText.value;
            if (!textToCopy) return;

            // Sélection et copie sécurisée
            outputText.select();
            try {
                const successful = document.execCommand('copy');
                if (successful) {
                    showToast("Copié dans le presse-papier !");
                    // Feedback visuel du bouton
                    copyBtn.classList.remove('bg-accent', 'hover:bg-accentHover');
                    copyBtn.classList.add('bg-emerald-600');
                    copyBtnText.textContent = "Copié !";
                    setTimeout(() => {
                        copyBtn.classList.remove('bg-emerald-600');
                        copyBtn.classList.add('bg-accent', 'hover:bg-accentHover');
                        copyBtnText.textContent = "Copier le texte";
                    }, 2000);
                } else {
                    showToast("Erreur lors de la copie");
                }
            } catch (err) {
                showToast("Impossible de copier automatiquement");
            }
        });

        // Gestion des boutons de pré-remplissage
        presetBtns.forEach(btn => {
            btn.addEventListener('click', () => {
                inputText.value = btn.textContent;
                // Déclencher l'événement input manuellement
                inputText.dispatchEvent(new Event('input'));
            });
        });

        // Fonction d'affichage du toast de notification
        function showToast(message) {
            const toastMessage = document.getElementById('toastMessage');
            toastMessage.textContent = message;
            toast.classList.remove('translate-y-20', 'opacity-0');
            toast.classList.add('translate-y-0', 'opacity-100');

            setTimeout(() => {
                toast.classList.remove('translate-y-0', 'opacity-100');
                toast.classList.add('translate-y-20', 'opacity-0');
            }, 2500);
        }

        // Valeur initiale par défaut pour illustrer directement
        window.addEventListener('DOMContentLoaded', () => {
            inputText.value = "Bonjour tout le monde ! 123";
            inputText.dispatchEvent(new Event('input'));
        });
